\section{Numerical Examples}
\subsection{A SDOF System with First Order Formulation}
We consider a SDOF system with natural frequency $\omega_n$ and damping parameters $m$ and $s$.
The system undergoes free vibration with initial conditions $u(0)=1$ and $\dot{u}(0)=0$.
\begin{gather}
    \left\{\begin{array}{l}
        \ddot{u}+\omega_n^2u+f=0, \\
        \dot{f}=-sf+m\omega^2u.
    \end{array}\right.
\end{gather}
The following parameters are used in the numerical simulation: $\omega_n=10$, $m=-2$, $s=10$.

\figref{fig:sdof} shows the displacement history of the SDOF system with various time step sizes $\Delta{}t$.
It is clear that second-order convergence of the used Newmark method is retained.

\subsection{A SDOF System with Second Order Formulation}
We consider a SDOF system with natural frequency $\omega_n$ and damping parameters $m$ and $s$.
The system undergoes free vibration with initial conditions $u(0)=1$ and $\dot{u}(0)=0$.
\begin{gather}
    \left\{\begin{array}{l}
        \ddot{u}+\omega_n^2u+f=0, \\
        \ddot{f}=s^2f+m\omega^2_n\dot{u}.
    \end{array}\right.
\end{gather}
The characteristic equation can be derived as
\begin{gather}
\ddddot{u}+\left(\omega_n^2-s^2\right)\ddot{u}+m\omega^2_n\dot{u}-s^2\omega_n^2u=0.
\end{gather}
\begin{gather}
\lambda^4+\left(\omega_n^2-s^2\right)\lambda^2+m\omega^2_n\lambda-s^2\omega_n^2=0.
\end{gather}
This quartic function can be decomposed into the product of two quadratic functions:
\begin{gather}
\lambda^4+\left(\omega_n^2-s^2\right)\lambda^2+m\omega^2_n\lambda-s^2\omega_n^2=\left(\lambda^2+a\lambda+b\right)\left(\lambda^2-a\lambda+c\right),
\end{gather}
providing that
\begin{gather}
b+c-a^2=\omega_n^2-s^2,\qquad
ac-ab=m\omega^2_n,\qquad
bc=-s^2\omega_n^2.
\end{gather}
